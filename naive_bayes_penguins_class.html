Story

We’ll start our naive Bayes classification with just a single penguin. Suppose an Antarctic researcher comes across a penguin that weighs less than 4200g with a 195mm-long flipper and 50mm-long bill. Our goal is to help this researcher identify the species of this penguin, Adelie, Chinstrap, or Gentoo.

Naive Bayes Classification!

naive_model_hints <- naiveBayes(species ~ flipper_length_mm + bill_length_mm + above_average_weight, data = penguins)

Our penguin

our_penguin <- data.frame(bill_length_mm = 50, flipper_length_mm = 195, above_average_weight = "no")

Ask our model to make a prediction of what species our penguin is

predict(naive_model_hints, newdata = our_penguin, type = "raw")

library(tidyverse)

# Build both curves on the SAME x-grid per species/feature, so the gap
# between "actual" and "assumed Normal" can be shaded directly rather
# than just visually estimated from two independently-drawn curves.
build_comparison <- function(values, species_name, feature_name, n_grid = 200) {
  values <- values[!is.na(values)]
  x_seq  <- seq(min(values), max(values), length.out = n_grid)
  
  actual_dens  <- density(values, n = n_grid, from = min(x_seq), to = max(x_seq))$y
  assumed_dens <- dnorm(x_seq, mean(values), sd(values))
  
  tibble(value = x_seq, actual = actual_dens, assumed = assumed_dens,
         species = species_name, feature = feature_name)
}

raw_data <- penguins %>%
  filter(!is.na(species)) %>%
  dplyr::select(species, bill_length_mm, flipper_length_mm) %>%
  pivot_longer(cols = c(bill_length_mm, flipper_length_mm),
               names_to = "feature", values_to = "value") %>%
  filter(!is.na(value))

comparison_data <- raw_data %>%
  group_by(species, feature) %>%
  summarise(data = list(build_comparison(value, unique(species), unique(feature))), .groups = "drop") %>%
  pull(data) %>%
  bind_rows()

feature_labels <- c(bill_length_mm = "Bill Length (mm)", flipper_length_mm = "Flipper Length (mm)")

ggplot(comparison_data, aes(x = value)) +
  geom_ribbon(aes(ymin = pmin(actual, assumed), ymax = pmax(actual, assumed)),
              fill = "grey55", alpha = 0.35) +
  geom_line(aes(y = actual, color = "Actual data"), linewidth = 1) +
  geom_line(aes(y = assumed, color = "Naive Bayes' assumed Normal"),
            linewidth = 1, linetype = "dashed") +
  geom_rug(data = raw_data, aes(x = value), sides = "b", alpha = 0.3, inherit.aes = FALSE) +
  facet_grid(feature ~ species, scales = "free",
             labeller = labeller(feature = feature_labels)) +
  scale_color_manual(values = c("Actual data" = "#4C72B0",
                                "Naive Bayes' assumed Normal" = "#DD8452"),
                     name = NULL) +
  labs(title = "How well does naive Bayes' Normal assumption fit the real data?",
       subtitle = "Shaded grey = the gap between what's real (blue) and what the model assumes (orange, dashed)",
       x = NULL, y = "Density") +
  theme_minimal(base_size = 13) +
  theme(legend.position = "top", strip.text = element_text(face = "bold"))
